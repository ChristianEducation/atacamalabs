import React from 'react';
import { AbsoluteFill, Easing, Img, OffthreadVideo, Sequence, interpolate, staticFile, useCurrentFrame, useVideoConfig } from 'remotion';
import { C, DM, MONO, SERIF } from './fonts';

export const ASTER_FPS = 30;
export const ASTER_DURATION = 1350; // 45 s
const clampOpts = { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' } as const;
const ease = Easing.bezier(0.22, 1, 0.36, 1);
const p = (f: number, a: number, b: number) => interpolate(f, [a, b], [0, 1], { ...clampOpts, easing: ease });

/** Entrada/salida suave de cada escena (superpuestas 8 fotogramas → fundido encadenado). */
const Scene: React.FC<{ children: React.ReactNode; fade?: number }> = ({ children, fade = 8 }) => {
  const f = useCurrentFrame();
  const { durationInFrames } = useVideoConfig();
  const o = Math.min(interpolate(f, [0, fade], [0, 1], clampOpts), interpolate(f, [durationInFrames - fade, durationInFrames], [1, 0], clampOpts));
  return <AbsoluteFill style={{ opacity: o }}>{children}</AbsoluteFill>;
};

/** Plano de video con deriva lenta de cámara y gradación unificada. */
const Plate: React.FC<{ src: string; drift?: [number, number]; dim?: number; blur?: number }> = ({ src, drift = [1.03, 1.1], dim = 0, blur = 0 }) => {
  const f = useCurrentFrame();
  const { durationInFrames } = useVideoConfig();
  const s = interpolate(f, [0, durationInFrames], drift, clampOpts);
  return (
    <AbsoluteFill style={{ overflow: 'hidden', background: C.dark }}>
      <OffthreadVideo src={staticFile(src)} muted style={{ width: '100%', height: '100%', objectFit: 'cover', transform: `scale(${s})`, filter: blur ? `blur(${blur}px)` : undefined }} />
      <AbsoluteFill style={{ background: 'linear-gradient(to top, rgba(4,18,40,0.78) 0%, rgba(4,18,40,0.28) 34%, rgba(4,18,40,0) 58%), linear-gradient(to bottom, rgba(4,18,40,0.62) 0%, rgba(4,18,40,0.2) 18%, rgba(4,18,40,0) 30%)' }} />
      {dim > 0 ? <AbsoluteFill style={{ background: `rgba(4,18,40,${dim})` }} /> : null}
    </AbsoluteFill>
  );
};

const Eyebrow: React.FC<{ text: string; at: number; color?: string; style?: React.CSSProperties }> = ({ text, at, color = '#ffffff', style }) => {
  const f = useCurrentFrame();
  const o = p(f, at, at + 14);
  return (
    <div style={{ display: 'flex', alignItems: 'center', gap: 14, opacity: o, transform: `translateY(${(1 - o) * 12}px)`, fontFamily: MONO, fontSize: 30, letterSpacing: '0.12em', textTransform: 'uppercase', color, textShadow: '0 2px 16px rgba(4,18,40,0.85)', ...style }}>
      <span style={{ width: 12, height: 12, borderRadius: 99, background: C.blue, boxShadow: '0 0 0 8px rgba(15,92,237,0.22)' }} />
      {text}
    </div>
  );
};

/** Titular serif palabra por palabra (misma jerarquía que el hero del sitio). */
const Words: React.FC<{ text: string; at: number; size?: number; step?: number; accent?: number[]; color?: string; style?: React.CSSProperties }> = ({ text, at, size = 92, step = 6, accent = [], color = '#fff', style }) => {
  const f = useCurrentFrame();
  return (
    <div style={{ fontFamily: SERIF, fontWeight: 500, fontSize: size, lineHeight: 1.04, letterSpacing: '-0.03em', color, textShadow: '0 4px 28px rgba(4,18,40,0.55)', ...style }}>
      {text.split(' ').map((w, i) => {
        const o = p(f, at + i * step, at + i * step + 16);
        return (
          <span key={i} style={{ display: 'inline-block', marginRight: '0.26em', opacity: o, transform: `translateY(${(1 - o) * 26}px)`, color: accent.includes(i) ? C.blueOnDark : color, fontStyle: accent.includes(i) ? 'italic' : 'normal' }}>
            {w}
          </span>
        );
      })}
    </div>
  );
};

/** Onda de audio determinista: amplitud según si la persona está hablando. */
const Wave: React.FC<{ speakFrom: number; speakTo: number; bars?: number }> = ({ speakFrom, speakTo, bars = 37 }) => {
  const f = useCurrentFrame();
  const amp = Math.min(p(f, speakFrom, speakFrom + 14), 1 - p(f, speakTo, speakTo + 16));
  return (
    <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'center', gap: 9, height: 150 }}>
      {Array.from({ length: bars }).map((_, i) => {
        const env = Math.abs(Math.sin(f * 0.17 + i * 0.62)) * (0.55 + 0.45 * Math.sin(i * 1.31 + f * 0.052));
        const mid = 1 - Math.abs(i - bars / 2) / (bars / 2);
        const h = 8 + amp * (14 + env * 120 * (0.35 + 0.65 * mid));
        return <div key={i} style={{ width: 9, height: h, borderRadius: 6, background: i % 6 === 0 ? '#fff' : C.blueOnDark, opacity: 0.55 + amp * 0.45 }} />;
      })}
    </div>
  );
};

};
