import { continueRender, delayRender, staticFile } from 'remotion';
import { DM as DM_FAMILY } from '../fonts'; // DM Sans (marca) — también dispara su carga

// Misma pareja tipográfica que atacamalabs.cl: Newsreader (títulos) + DM Sans (UI y cuerpo).
export const DM = DM_FAMILY;
export const SERIF = '"Newsreader", Georgia, serif';
export const MONO = 'ui-monospace, SFMono-Regular, Consolas, monospace';

const h = delayRender('Newsreader');
Promise.all([
  new FontFace('Newsreader', `url(${staticFile('fonts/Newsreader-latin-wght.woff2')}) format("woff2")`, { weight: '200 800', style: 'normal' }).load(),
  new FontFace('Newsreader', `url(${staticFile('fonts/Newsreader-latin-wght-italic.woff2')}) format("woff2")`, { weight: '200 800', style: 'italic' }).load(),
])
  .then((faces) => {
    faces.forEach((ff) => (document.fonts as unknown as { add(f: FontFace): void }).add(ff));
    continueRender(h);
  })
  .catch(() => continueRender(h));
